# 后端设计（backend/）

```text
backend/
├── pom.xml
└── src/
    └── main/
        ├── java/
        │   └── com/yourname/loganomaly/
        │       ├── LogAnomalyApplication.java          # 启动类
        │       │
        │       ├── config/                             # 配置
        │       │   ├── WebConfig.java                  # 跨域、拦截器
        │       │   ├── RedisConfig.java                # Redis 序列化配置
        │       │   └── MybatisPlusConfig.java          # 分页插件
        │       │
        │       ├── common/                             # 公共模块
        │       │   ├── Result.java                     # 统一返回体
        │       │   ├── PageResult.java                 # 分页返回体
        │       │   ├── exception/
        │       │   │   ├── BusinessException.java
        │       │   │   └── GlobalExceptionHandler.java
        │       │   └── utils/
        │       │       ├── JwtUtil.java
        │       │       ├── IpUtil.java                 # 获取真实IP、校验IP、判断内网
        │       │       └── CsvUtil.java                # CSV解析与脏数据处理
        │       │
        │       ├── controller/                         # 接口层
        │       │   ├── AuthController.java             # 登录
        │       │   ├── UploadController.java           # CSV上传
        │       │   ├── LogController.java              # 日志查询/分页/导出
        │       │   ├── DetectController.java           # 触发检测
        │       │   ├── AlertController.java            # 告警管理
        │       │   └── StatsController.java            # 可视化统计
        │       │
        │       ├── service/                            # 业务层
        │       │   ├── AuthService.java
        │       │   ├── UploadService.java              # 解析+脏数据处理+入库
        │       │   ├── LogService.java
        │       │   ├── DetectService.java              # 调Python + 规则检测 + 合并
        │       │   ├── AlertService.java
        │       │   ├── StatsService.java
        │       │   └── IpLocationService.java          # IP归属地 + Redis缓存
        │       │
        │       ├── mapper/                             # MyBatis-Plus Mapper
        │       │   ├── AccessLogMapper.java
        │       │   ├── AlertMapper.java
        │       │   ├── UserMapper.java
        │       │   └── UploadTaskMapper.java
        │       │
        │       ├── entity/                             # 数据库实体
        │       │   ├── AccessLog.java
        │       │   ├── Alert.java
        │       │   ├── User.java
        │       │   └── UploadTask.java
        │       │
        │       ├── dto/                                # 请求/响应对象
        │       │   ├── LoginDTO.java
        │       │   ├── LogQueryDTO.java
        │       │   ├── DetectRequestDTO.java
        │       │   └── DetectResultDTO.java
        │       │
        │       ├── vo/                                 # 前端展示对象
        │       │   ├── LogVO.java
        │       │   ├── AlertVO.java
        │       │   ├── TrendVO.java
        │       │   ├── TopIpVO.java
        │       │   └── IpLocationVO.java
        │       │
        │       ├── rule/                               # 规则检测引擎
        │       │   ├── RuleEngine.java
        │       │   ├── Rule.java
        │       │   └── impl/
        │       │       ├── HighFrequencyRule.java
        │       │       ├── ScanRule.java
        │       │       └── SlowResponseRule.java
        │       │
        │       └── client/                             # 调Python的客户端
        │           └── PythonDetectClient.java
        │
        └── resources/
            ├── application.yml
            ├── mapper/                                 # XML（复杂SQL）
            │   ├── AccessLogMapper.xml
            │   └── StatsMapper.xml
            └── sql/
                └── init.sql                            # 建表语句
```
